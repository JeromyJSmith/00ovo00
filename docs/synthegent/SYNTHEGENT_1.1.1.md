Copydef detect_isomorphisms(findings: List[Finding]) -> List[Isomorphism]:
    """
    Find structural similarities across domain-distinct findings.
    Uses graph kernel similarity (Weisfeiler-Leman) + embedding cosine.
    """
    # Convert each finding to relational graph
    graphs = [finding_to_graph(f) for f in findings]

    # Compute pairwise similarity
    isomorphisms = []
    for i, g1 in enumerate(graphs):
        for j, g2 in enumerate(graphs[i+1:], start=i+1):
            if findings[i].domain == findings[j].domain:
                continue  # Skip same-domain comparisons
            
            structural_sim = weisfeiler_leman_similarity(g1, g2)
            semantic_sim = cosine_similarity(
                findings[i].embedding,
                findings[j].embedding
            )
            
            if structural_sim > 0.75 and semantic_sim > 0.60:
                # Likely isomorphism
                isomorphisms.append(Isomorphism(
                    finding_a=findings[i],
                    finding_b=findings[j],
                    structural_score=structural_sim,
                    semantic_score=semantic_sim,
                    abstraction=extract_abstract_pattern(g1, g2)
                ))
    
    return isomorphisms

class GraphAugmentedGeneration:
    """
    Replaces traditional RAG with graph-aware retrieval.
    NO RE-RANKING. ALL matches go into the graph.
    """

    def __init__(self, neo4j_client, vector_db):
        self.graph = neo4j_client
        self.vectors = vector_db  # ChromaDB
        
    def retrieve_and_graph(self, query: str, threshold=0.3):
        """
        Step 1: Get ALL semantic matches above threshold.
        Step 2: Add to graph with relationships.
        Step 3: Return graph subgraph, not ranked list.
        """
        # Vector search (no top-k limit!)
        all_matches = self.vectors.query(
            query_embedding=self.embed(query),
            n_results=1000,              # Large limit
            where={"similarity": {"$gte": threshold}}
        )
        
        # Add nodes to graph
        for match in all_matches:
            self.graph.merge_node(
                label="ResearchOutput",
                properties={
                    "output_id": match["id"],
                    "content": match["text"],
                    "embedding": match["embedding"],
                    "similarity_to_query": match["similarity"]
                }
            )
            
            # Link to source node
            self.graph.create_relationship(
                source_node=match["source_node_id"],
                target_node=match["id"],
                rel_type="GENERATED"
            )
        
        # Compute semantic relationships between matches
        self.compute_semantic_edges(all_matches)
        
        # Return graph query interface, not documents
        return self.graph.subgraph(
            anchor_query=query,
            max_hops=3  # Traverse 3 degrees of separation
        )
    
    def compute_semantic_edges(self, nodes: List[dict]):
        """
        Find relationships between retrieved nodes.
        This is where magic happens: contradictions, support, isomorphisms.
        """
        for i, node_a in enumerate(nodes):
            for node_b in nodes[i+1:]:
                similarity = cosine_similarity(
                    node_a["embedding"], 
                    node_b["embedding"]
                )
                
                if similarity > 0.8:
                    self.graph.create_relationship(
                        source_node=node_a["id"],
                        target_node=node_b["id"],
                        rel_type="SUPPORTS",
                        properties={"similarity": similarity}
                    )
                elif self.is_contradiction(node_a["content"], node_b["content"]):
                    self.graph.create_relationship(
                        source_node=node_a["id"],
                        target_node=node_b["id"],
                        rel_type="CONTRADICTS",
                        properties={"severity": self.contradiction_score(node_a, node_b)}
                    )

class GeosemanticAnalyzer:
    """
    Answers: "Do researchers in different locations
    converge on the same insights?"
    """

    def detect_distributed_consensus(self, topic: str):
        """
        Query: Find claims about {topic} from geographically 
        distant nodes with high semantic similarity.
        """
        query = f"""
        MATCH (n1:NetworkNode)-[:GENERATED]->(out1:ResearchOutput)
              -[:CONTAINS]->(c1:Claim)
        MATCH (n2:NetworkNode)-[:GENERATED]->(out2:ResearchOutput)
              -[:CONTAINS]->(c2:Claim)
        WHERE n1.node_id <> n2.node_id
          AND point.distance(n1.location, n2.location) > 1000000  // >1000km apart
          AND c1.text CONTAINS '{topic}'
          AND c2.text CONTAINS '{topic}'
          AND gds.similarity.cosine(c1.embedding, c2.embedding) > 0.85
        RETURN 
          n1.location as location_1,
          n2.location as location_2,
          c1.text as claim_1,
          c2.text as claim_2,
          gds.similarity.cosine(c1.embedding, c2.embedding) as similarity
        ORDER BY similarity DESC
        """
        
        results = self.graph.query(query)
        
        # Interpretation
        if len(results) > 10:
            return {
                "status": "DISTRIBUTED_CONSENSUS",
                "evidence": results,
                "interpretation": f"Found {len(results)} geographically independent researchers converging on similar insights about {topic}. This suggests robust truth signal."
            }
        else:
            return {
                "status": "LOCALIZED_INSIGHT",
                "interpretation": f"Insight about {topic} may be geographically or culturally specific."
